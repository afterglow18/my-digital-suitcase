3 — Create src/lib/backgroundRemoval.ts

import { removeBackground as imglyRemoveBackground } from "@imgly/background-removal";
/**
 * Remove the background from a JPEG/PNG base64 data-URL.
 * Returns a PNG data-URL with transparent background.
 * On first ever call downloads ~15 MB ONNX model from imgly CDN (cached after that).
 * Throws on network error or unreadable image — callers should catch and fall back.
 */
export async function removeBackground(dataUrl: string): Promise<string> {
  const sourceBlob = await dataUrlToBlob(dataUrl);
  const resultBlob = await imglyRemoveBackground(sourceBlob, {
    model: "isnet_fp16",  // valid: "isnet" | "isnet_fp16" | "isnet_quint8" — NOT "small"/"medium"
    output: { format: "image/png", quality: 0.9 },
    // publicPath omitted → uses staticimgly CDN automatically
  });
  return blobToDataUrl(resultBlob);
}
export function blobToDataUrl(blob: Blob): Promise<string> {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload  = () => resolve(reader.result as string);
    reader.onerror = () => reject(new Error("FileReader failed"));
    reader.readAsDataURL(blob);
  });
}
export async function dataUrlToBlob(dataUrl: string): Promise<Blob> {
  const res = await fetch(dataUrl);
  return res.blob();
}

4c — encodeForUpload (put outside the component)

async function encodeForUpload(input: File | Blob): Promise<Blob> {
  return new Promise((resolve, reject) => {
    const objectUrl = URL.createObjectURL(input);
    const img = new Image();
    img.onload = () => {
      URL.revokeObjectURL(objectUrl);
      const MAX = 2048;
      const scale = Math.min(1, MAX / Math.max(img.naturalWidth, img.naturalHeight));
      const w = Math.round(img.naturalWidth  * scale);
      const h = Math.round(img.naturalHeight * scale);
      const canvas = document.createElement("canvas");
      canvas.width = w;
      canvas.height = h;
      canvas.getContext("2d")!.drawImage(img, 0, 0, w, h);
      canvas.toBlob(
        (b) => (b && b.size > 1000 ? resolve(b) : reject(new Error("blank image"))),
        "image/jpeg",
        0.85,
      );
    };
    img.onerror = () => {
      URL.revokeObjectURL(objectUrl);
      reject(new Error("failed to load image"));
    };
    img.src = objectUrl;
  });
}

4e — handleFile (call after user picks or takes a photo)

const handleFile = useCallback(async (file: File | Blob) => {
  setErrorMsg(null);
  // Switch to "encoding" phase BEFORE any async work so the user sees a
  // full-screen spinner immediately instead of a blank pick screen for 1-3 s.
  const myGen = ++bgGenRef.current;
  setOriginalBlob(null);
  setOriginalUrl(null);
  setCleanedBlob(null);
  setCleanedUrl(null);
  setBgFailed(false);
  setBgProcessing(false);
  setSelected("original");
  setPhase("encoding");
  // Encode to JPEG ≤ 2048px
  let jpeg: Blob;
  try {
    jpeg = await encodeForUpload(file);
  } catch (err) {
    if (bgGenRef.current !== myGen) return;
    setErrorMsg(`Could not read the photo: ${err instanceof Error ? err.message : String(err)}`);
    setPhase("pick");
    return;
  }
  if (bgGenRef.current !== myGen) return;
  // Show original, switch to comparison screen
  setOriginalBlob(jpeg);
  setOriginalUrl(URL.createObjectURL(jpeg));
  setPhase("preview");
  // Background removal — generation guard discards stale results
  setBgProcessing(true);
  try {
    const dataUrl = await blobToDataUrl(jpeg);
    if (bgGenRef.current !== myGen) return;
    const resultUrl = await removeBackground(dataUrl);
    if (bgGenRef.current !== myGen) return;
    const resultBlob   = await dataUrlToBlob(resultUrl);
    const resultObjUrl = URL.createObjectURL(resultBlob);
    if (bgGenRef.current !== myGen) { URL.revokeObjectURL(resultObjUrl); return; }
    setCleanedBlob(resultBlob);
    setCleanedUrl(resultObjUrl);
    setSelected("cleaned");
  } catch (err) {
    if (bgGenRef.current !== myGen) return;
    console.warn("Background removal failed:", err);
    setBgFailed(true);
  } finally {
    if (bgGenRef.current === myGen) setBgProcessing(false);
  }
}, []);

Bugs you will hit without these exact patterns

Bug	What happens without the fix
Generation counter (bgGenRef)	Take photo #2 while #1 is still processing → #1's finally block fires setBgProcessing(false) on #2's state, killing the spinner silently. #1's result also overwrites #2's cleaned image.
setBgProcessing(false) in handleClose	Close the sheet mid-removal → bgProcessing stays true. Next open has Save disabled with no explanation.
phase = "encoding" before first await	Skip it → user sits on the pick screen for 1–3 s during canvas encode. Looks like a crash.
No AnimatePresence on phase switching	Any AnimatePresence wrapper around {phase === "x" && ...} blocks creates exit-animation windows where no child is mounted — blank screen between every phase change, regardless of mode, initial, or transition duration. Use plain divs.
Model name "isnet_fp16" not "medium"	"medium" is not valid in v1.7 → runtime error. Valid values: "isnet", "isnet_fp16", "isnet_quint8".
optimizeDeps: { exclude: ['@imgly/background-removal'] }	Skip it → Vite crashes pre-bundling the WASM imports.
First use downloads ~15 MB	The ONNX model downloads once from imgly CDN then WKWebView/browser caches it permanently. Show "This will take a moment…" so users don't think it froze.
How did we fix the 1-2 second balnknscreen
Three separate fixes stacked on top of each other:

1. The root cause — AnimatePresence mode="wait" Every phase change (pick → encoding → preview) makes the exiting element fade out before the entering element starts. With the default spring animation that exit takes ~300–400 ms. With three phase changes in quick succession you got three gaps back to back — easily 1 second of nothing on screen. Fix: removed AnimatePresence entirely from phase switching and used plain conditional divs. Content now switches instantly, no gap possible.

2. The initial open blank — initial={false} was missing Even on the very first render when the sheet opened, the pick screen started at opacity: 0 and animated to opacity: 1 over ~400 ms. Looked blank every time the sheet opened. Fix was initial={false} on AnimatePresence — but since we removed AnimatePresence completely this is now a non-issue.

3. The pick screen delay — setPhase("encoding") before any await encodeForUpload (canvas resize + toBlob) takes 1–3 seconds on device. Previously setPhase("preview") was only called after encoding finished, so the user sat on the pick screen for 1–3 seconds with no feedback. Fix: call setPhase("encoding") as the very first thing in handleFile, before any await, so the "Processing…" full-screen spinner appears instantly the moment the user picks a photo.
